<span class="line"><span style="color:#A0111F;--shiki-dark:#FF9492">    if</span><span style="color:#0E1116;--shiki-dark:#F0F3F6"> (</span><span style="color:#A0111F;--shiki-dark:#FF9492">!</span><span style="color:#0E1116;--shiki-dark:#F0F3F6">(key </span><span style="color:#A0111F;--shiki-dark:#FF9492">in</span><span style="color:#0E1116;--shiki-dark:#F0F3F6"> vm)) {</span></span>
<span class="line"><span style="color:#622CBC;--shiki-dark:#DBB7FF">      defineComputed</span><span style="color:#0E1116;--shiki-dark:#F0F3F6">(vm, key, userDef)</span></span>
<span class="line"><span style="color:#0E1116;--shiki-dark:#F0F3F6">    } </span><span style="color:#A0111F;--shiki-dark:#FF9492">else</span><span style="color:#A0111F;--shiki-dark:#FF9492"> if</span><span style="color:#0E1116;--shiki-dark:#F0F3F6"> (process.env.</span><span style="color:#023B95;--shiki-dark:#91CBFF">NODE_ENV</span><span style="color:#A0111F;--shiki-dark:#FF9492"> !==</span><span style="color:#032563;--shiki-dark:#ADDCFF"> 'production'</span><span style="color:#0E1116;--shiki-dark:#F0F3F6">) {</span></span>
<span class="line"><span style="color:#A0111F;--shiki-dark:#FF9492">      if</span><span style="color:#0E1116;--shiki-dark:#F0F3F6"> (key </span><span style="color:#A0111F;--shiki-dark:#FF9492">in</span><span style="color:#0E1116;--shiki-dark:#F0F3F6"> vm.$data) {</span></span>

<span class="line"><span style="color:#0E1116;--shiki-dark:#F0F3F6">      }</span></span>
